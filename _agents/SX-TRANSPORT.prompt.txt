```rust
fn calculate_retry_delay(retry_count: u8) -> Duration {
    let base = 2_u64.pow(retry_count as u32);
    let jitter = (rand::random::<f64>() * 0.4) - 0.2; // ±20%
    let seconds = (base as f64 * (1.0 + jitter)).min(30.0);
    Duration::from_secs(seconds as u64)
}
```
